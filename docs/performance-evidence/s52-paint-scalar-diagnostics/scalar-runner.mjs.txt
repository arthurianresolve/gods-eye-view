const stage = path.dirname(fileURLToPath(import.meta.url));
const root = path.resolve(stage, '..', '..');
const sourceRoot = path.join(stage, 'named-entry-capture-controls-01', 'submarine-cables-named-entry');

const workerHash = '9dc24d3f9b017d19df6ca89270803c01bf811b1c93896e3495b27a4d16aff901';
const sha = (bytes) => createHash('sha256').update(bytes).digest('hex');

if (!outArg) throw new Error('Usage: run-scalar-snapshot-storage-controls.mjs fresh-output-directory');

const runtimePath = path.join(sourceRoot, 'src/overlays/worldOverlay.js');
const reference = readFileSync(runtimePath, 'utf8');
assert.equal(sha(Buffer.from(reference)), expectedSource);

function replaceOnce(text, before, after, label) {
  const at = text.indexOf(before);
  assert.ok(at >= 0, `Missing ${label}`);
  assert.equal(text.indexOf(before, at + before.length), -1, `Ambiguous ${label}`);
  return text.slice(0, at) + after + text.slice(at + before.length);
}

mkdirSync(out, { recursive: true });
const workloads = [
  { profile: 'local-infrastructure', entries: 320, order: ['reference', 'candidate'] },
  { profile: 'submarine-cables', entries: 160, order: ['candidate', 'reference'] },
];
const parityKeys = ['candidateCount', 'paintedCount', 'paintedBySource', 'solveCount'];
const results = {};
let order = 0;
for (const workload of workloads) {
  const pair = {};
  for (const variant of workload.order) {
    const name = `${workload.profile}-${variant}`;
    const target = path.join(out, name);
    cpSync(sourceRoot, target, { recursive: true });
    const targetRuntime = path.join(target, 'src/overlays/worldOverlay.js');
    writeFileSync(targetRuntime, variant === 'candidate' ? candidate : reference, { flag: 'w' });
    const workerPath = path.join(target, 'src/overlays/worldOverlayAllocation.worker.mjs');
    assert.equal(sha(readFileSync(workerPath)), workerHash);
    const syntax = spawnSync(process.execPath, ['--check', targetRuntime], { encoding: 'utf8', timeout: 15_000 });
    if (syntax.error) throw syntax.error;
    assert.equal(syntax.status, 0, `${name} syntax: ${syntax.stderr}`);
    const worker = spawnSync(process.execPath, ['--expose-gc', '--no-concurrent-recompilation', workerPath], {
      cwd: target, encoding: 'utf8', timeout: 240_000, maxBuffer: 8 * 1024 * 1024,
      env: { ...process.env, GEV_ALLOC_WARMUP: '3000', GEV_ALLOC_CHUNK: '48', GEV_ALLOC_CHUNKS: '10', GEV_ALLOC_ENTRIES: String(workload.entries), GEV_ALLOC_PROFILE: workload.profile },
    });
    const stdout = worker.stdout || '';
    const stderr = worker.stderr || '';
    writeFileSync(path.join(target, 'worker.stdout.txt'), stdout, { flag: 'w' });
    writeFileSync(path.join(target, 'worker.stderr.txt'), stderr, { flag: 'w' });
    if (worker.error) throw worker.error;
    assert.equal(worker.status, 0, `${name} worker: ${stderr}`);
    const measured = JSON.parse(stdout.trim().split(/\r?\n/).at(-1));
    assert.equal(measured.ok, true);
    assert.equal(measured.profile, workload.profile);
    assert.equal(measured.entryCount, workload.entries);
    pair[variant] = { order: order++, runtimeSha256: sha(readFileSync(targetRuntime)), stdoutSha256: sha(Buffer.from(stdout)), stderrSha256: sha(Buffer.from(stderr)), measured };
    results[name] = pair[variant];
  }
  for (const key of parityKeys) assert.deepEqual(pair.reference.measured[key], pair.candidate.measured[key], `${workload.profile} parity: ${key}`);
}
