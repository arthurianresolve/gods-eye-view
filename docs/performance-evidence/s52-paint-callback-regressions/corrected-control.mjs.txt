import assert from 'node:assert/strict';
import { readFileSync, writeFileSync, cpSync, mkdirSync } from 'node:fs';
import { spawnSync } from 'node:child_process';
import { createHash } from 'node:crypto';
const source = 'qa-artifacts/s52-paint-stage-rebased-936bd/didpaint-signature-prototype-02/candidate';
const output = 'qa-artifacts/s52-corrected-paint-review-01';
const hash = b => createHash('sha256').update(b).digest('hex');
const runtime = readFileSync(`${source}/src/overlays/worldOverlay.js`);
assert.equal(hash(runtime), '2cf2d2c0cbaa22e17e6b32736c461c8a05331129bc7809ceeea5d680bfa1c523');
const appendix = readFileSync('qa-artifacts/s52-self-unregister-review-02/regression.append.mjs.txt');
assert.equal(hash(appendix), 'e8b4b4e0f5ad7e68c804e3a8880938c7cca5ea3bf945a997815df8423b524bb8');
mkdirSync(output);
cpSync(source, `${output}/candidate`, { recursive: true });
const test = `${output}/candidate/src/overlays/worldOverlay.test.mjs`;
writeFileSync(test, Buffer.concat([readFileSync(test), appendix]));
const rows = [];
for (const [name, paths, count, extra] of [
  ['corrected-prototype', [test], 2, ['--test-name-pattern=root paint cache']],
  ['production-focused', ['src/overlays/worldOverlay.test.mjs', 'src/overlays/worldOverlayDraw.test.mjs', 'src/overlays/worldOverlayFontWake.test.mjs', 'src/overlays/worldOverlayTokens.test.mjs'], 90, []],
]) {
  const result = spawnSync(process.execPath, ['--test', '--test-reporter=tap', '--test-concurrency=1', ...extra, ...paths], { timeout: 90_000, maxBuffer: 2e7 });
  if (result.error) throw result.error;
  for (const stream of ['stdout', 'stderr']) writeFileSync(`${output}/${name}.${stream}.txt`, result[stream], { flag: 'wx' });
  assert.equal(result.status, 0, result.stdout.toString() + result.stderr.toString());
  const counts = Object.fromEntries(['tests', 'pass', 'fail', 'skipped'].map(key => {
    const matches = [...result.stdout.toString().matchAll(new RegExp(`^# ${key} (\\d+)$`, 'gm'))];
    assert.equal(matches.length, 1, `Missing or duplicate ${key}`);
    return [key, Number(matches[0][1])];
  }));
  assert.deepEqual(counts, { tests: count, pass: count, fail: 0, skipped: 0 });
  rows.push({ name, ...counts, sourceSha256: hash(name === 'corrected-prototype' ? runtime : readFileSync('src/overlays/worldOverlay.js')), stdoutSha256: hash(result.stdout), stderrSha256: hash(result.stderr) });
}
const report = { schema: 'gev-corrected-paint-regression-review/v1', node: process.version, platform: process.platform, appendixSha256: hash(appendix), productionTestSha256: hash(readFileSync('src/overlays/worldOverlay.test.mjs')), scope: 'Independent unit failure controls and production focused checks; no native-pixel, GPU or performance claim.', rows };
writeFileSync(`${output}/review.json`, JSON.stringify(report, null, 2) + '\n', { flag: 'wx' });
console.log(JSON.stringify(report));
